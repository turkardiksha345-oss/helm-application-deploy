1. Check pods:
   kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

2. Check services:
   kubectl get svc -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

{{- if .Values.ingress.enabled }}
3. Check ingress:
   kubectl get ingress -n {{ .Release.Namespace }} {{ include "two-tier-app.fullname" . }}
{{- else }}
3. Port-forward the frontend service:
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "two-tier-app.frontendName" . }} 8080:{{ .Values.frontend.service.port }}

   Open http://localhost:8080
{{- end }}
